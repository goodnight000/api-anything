$ # Demo 2: learn an operation from two examples, then call it with new input. Live, no cuts.
$ ./examples/teach-once.sh arxiv.org
# 1. Learn from two examples. Headless Chrome opens the page twice. Summarized with jq.
$ api-anything add hn-demo links \
  --trigger 'https://news.ycombinator.com/from?site={domain}' \
  --example domain=github.com --example2 domain=github.io --match path=/from \
  --html '{"items":"tr.athing","fields":{"title":".titleline > a","url":".titleline > a@href"}}'

  "preview_first": "https://github.com/sevenDatabase/SevenDB"
}

# 2. The saved template has a slot where the domain goes.
$ jq -c '.operations[] | select(.name == "links").slots' "/tmp/api-anything-demo/home/sites/hn-demo.j
